# Core Matchmaking App Implementation Plan (Plan 1 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the working pickleball matchmaking app: create a session with courts and players, auto-assign games (fairness or simple-queue mode), finish games, handle breaks/late arrivals/court changes, undo, and keep a saved player roster on the device. Installable and usable offline.

**Architecture:** A pure TypeScript engine (`src/engine`) holds all matchmaking logic. A Zustand session store (`src/store`) wraps it with undo and `localStorage` persistence; the roster lives in IndexedDB. React screens (`src/ui`) only read the store and dispatch actions. The roster data shape already matches the spec (soft delete, `photoVersion`, `updatedAt`) so Plans 2 (photos) and 3 (Supabase sync) can extend it without migration of meaning.

**Tech Stack:** React 18, TypeScript, Vite 6, Zustand 5, idb, Vitest 3 + Testing Library + fake-indexeddb, vite-plugin-pwa.

**Spec:** `docs/superpowers/specs/2026-11-06-pickleball-matchmaking-design.md`

**Roadmap:** Plan 2 adds player photos (camera capture, crop/compress, avatars). Plan 3 adds Supabase auth + sync. Neither is part of this plan.

**Spec clarifications decided while planning (also applied to the spec):**
1. Player names must be unique among players who have **not left** (so a player who left can be re-added by name).
2. A player returning from a break starts at `max(own gamesPlayed, floor(average of other active players))`, so a break never *reduces* someone's games played.

```ts file=src/engine/rng.ts
/** Small deterministic PRNG (mulberry32). Returns numbers in [0, 1). */
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```ts file=src/engine/pool.ts
import { EngineError } from './errors';
import type { Player, PlayerId, Session } from './types';

export function playerById(s: Session, id: PlayerId): Player {
  const p = s.players.find((q) => q.id === id);
  if (!p) throw new EngineError('NOT_FOUND', `Player ${id} not found`);
  return p;
}

export function onCourtIds(s: Session): Set<PlayerId> {
  const ids = new Set<PlayerId>();
  for (const court of s.courts) {
    if (court.game) {
      for (const id of [...court.game.team1, ...court.game.team2]) ids.add(id);
    }
  }
  return ids;
}

/** Active players who are not currently on a court. */
export function availablePlayers(s: Session): Player[] {
  const busy = onCourtIds(s);
  return s.players.filter((p) => p.status === 'active' && !busy.has(p.id));
}

/** Fewest games first, then waited longest, then the stable random tie key. */
export function compareByPriority(a: Player, b: Player): number {
  return (
    a.gamesPlayed - b.gamesPlayed ||
    a.lastFinishedSeq - b.lastFinishedSeq ||
    a.tieKey - b.tieKey
  );
}

/**
 * Available players in the order they will be picked.
 * Fairness mode: by priority. Queue mode: FIFO by queueOrder.
 */
export function orderedWaiting(s: Session): Player[] {
  const pool = availablePlayers(s);
  if (s.mode === 'fairness') return [...pool].sort(compareByPriority);
  const byId = new Map(pool.map((p) => [p.id, p]));
  return s.queueOrder.flatMap((id) => {
    const p = byId.get(id);
    return p ? [p] : [];
  });
}

/** Floor of the average gamesPlayed of active players (optionally excluding one). */
export function lateArrivalGames(s: Session, excludeId?: PlayerId): number {
  const active = s.players.filter((p) => p.status === 'active' && p.id !== excludeId);
  if (active.length === 0) return 0;
  return Math.floor(active.reduce((sum, p) => sum + p.gamesPlayed, 0) / active.length);
}
```

```ts file=src/engine/selection.ts
import type { Player, Team } from './types';

type Pair = [Player, Player];

function splitCost(t1: Pair, t2: Pair): number {
  const partnerRepeats =
    (t1[0].partnerCounts[t1[1].id] ?? 0) + (t2[0].partnerCounts[t2[1].id] ?? 0);
  let opponentRepeats = 0;
  for (const a of t1) {
    for (const b of t2) opponentRepeats += a.opponentCounts[b.id] ?? 0;
  }
  return partnerRepeats * 10 + opponentRepeats;
}

/**
 * Of the 3 ways to split 4 players into two teams, choose the one with the
 * fewest repeat partners (weight 10) and repeat opponents (weight 1).
 * Exact ties are broken with the injected rng.
 */
export function bestSplit(four: Player[], rng: () => number): [Team, Team] {
  if (four.length !== 4) throw new Error('bestSplit needs exactly 4 players');
  const [a, b, c, d] = four;
  const options: [Pair, Pair][] = [
    [[a, b], [c, d]],
    [[a, c], [b, d]],
    [[a, d], [b, c]],
  ];
  const costs = options.map(([t1, t2]) => splitCost(t1, t2));
  const min = Math.min(...costs);
  const best = options.filter((_, i) => costs[i] === min);
  const [t1, t2] = best[Math.floor(rng() * best.length)];
  return [
    [t1[0].id, t1[1].id],
    [t2[0].id, t2[1].id],
  ];
}
```

```ts file=src/engine/fill.ts
import { orderedWaiting } from './pool';
import { bestSplit } from './selection';
import type { Ctx, Session, Team } from './types';

/**
 * Starts a game on every empty court that can be filled.
 * MUTATES `s` — callers must pass a clone.
 *
 * Both modes take the first 4 of `orderedWaiting`. Fairness mode then picks the
 * best team split; queue mode uses the order as-is (first two vs last two).
 */
export function fillCourts(s: Session, ctx: Ctx): void {
  const courts = [...s.courts].sort((a, b) => a.number - b.number);
  for (const court of courts) {
    if (court.status !== 'empty') continue;
    const four = orderedWaiting(s).slice(0, 4);
    if (four.length < 4) return;
    const ids = four.map((p) => p.id);
    const [team1, team2]: [Team, Team] =
      s.mode === 'queue'
        ? [
            [ids[0], ids[1]],
            [ids[2], ids[3]],
          ]
        : bestSplit(four, ctx.rng);
    court.status = 'playing';
    court.game = { team1, team2, startedAt: ctx.now };
    s.queueOrder = s.queueOrder.filter((id) => !ids.includes(id));
  }
}
```

```ts file=src/engine/session.ts
import { cloneSession } from './clone';
import { EngineError } from './errors';
import { fillCourts } from './fill';
import { lateArrivalGames } from './pool';
import type { Court, Ctx, NewPlayerInput, NewSessionInput, Player, Session } from './types';

const normalize = (name: string) => name.trim().toLowerCase();

function makeCourt(number: number, ctx: Ctx): Court {
  return { id: ctx.newId(), number, status: 'empty', closing: false };
}

/** MUTATES `s`. Validates, creates the player, and queues them at the back. */
function insertPlayer(s: Session, input: NewPlayerInput, ctx: Ctx): void {
  const name = input.name.trim();
  if (!name) throw new EngineError('EMPTY_NAME', 'Enter a name for the player.');
  if (s.players.some((p) => p.status !== 'left' && normalize(p.name) === normalize(name))) {
    throw new EngineError('DUPLICATE_NAME', `${name} is already in this session.`);
  }
  const player: Player = {
    id: ctx.newId(),
    name,
    rosterId: input.rosterId,
    status: 'active',
    gamesPlayed: lateArrivalGames(s),
    lastFinishedSeq: s.gameSeq,
    tieKey: ctx.rng(),
    partnerCounts: {},
    opponentCounts: {},
  };
  s.players.push(player);
  s.queueOrder.push(player.id);
}

export function createSession(input: NewSessionInput, ctx: Ctx): Session {
  if (!Number.isInteger(input.courtCount) || input.courtCount < 1) {
    throw new EngineError('INVALID_COURTS', 'A session needs at least one court.');
  }
  const s: Session = {
    id: ctx.newId(),
    name: input.name.trim() || 'Pickleball session',
    mode: input.mode,
    courts: Array.from({ length: input.courtCount }, (_, i) => makeCourt(i + 1, ctx)),
    players: [],
    queueOrder: [],
    history: [],
    gameSeq: 0,
    createdAt: ctx.now,
  };
  // Players are inserted first and courts filled once, so a whole group starts together.
  for (const p of input.players) insertPlayer(s, p, ctx);
  fillCourts(s, ctx);
  return s;
}

export function addPlayer(s: Session, input: NewPlayerInput, ctx: Ctx): Session {
  const next = cloneSession(s);
  insertPlayer(next, input, ctx);
  fillCourts(next, ctx);
  return next;
}

export function endSession(s: Session, ctx: Ctx): Session {
  const next = cloneSession(s);
  next.endedAt = ctx.now;
  return next;
}
```

```ts file=src/engine/game.ts
import { cloneSession } from './clone';
import { EngineError } from './errors';
import { fillCourts } from './fill';
import { playerById } from './pool';
import type { Ctx, PlayerId, Session } from './types';

function bump(counts: Record<PlayerId, number>, id: PlayerId): void {
  counts[id] = (counts[id] ?? 0) + 1;
}

export function finishGame(s: Session, courtId: string, ctx: Ctx): Session {
  const n = cloneSession(s);
  const court = n.courts.find((c) => c.id === courtId);
  if (!court) throw new EngineError('NOT_FOUND', 'Court not found.');
  if (court.status !== 'playing' || !court.game) {
    throw new EngineError('NOT_PLAYING', 'No game is being played on this court.');
  }

  const { team1, team2, startedAt } = court.game;
  const everyone = [...team1, ...team2];
  n.gameSeq += 1;

  for (const id of everyone) {
    const p = playerById(n, id);
    p.gamesPlayed += 1;
    p.lastFinishedSeq = n.gameSeq;
  }
  for (const [a, b] of [team1, team2]) {
    bump(playerById(n, a).partnerCounts, b);
    bump(playerById(n, b).partnerCounts, a);
  }
  for (const a of team1) {
    for (const b of team2) {
      bump(playerById(n, a).opponentCounts, b);
      bump(playerById(n, b).opponentCounts, a);
    }
  }

  n.history.push({ team1, team2, courtNumber: court.number, startedAt, finishedAt: ctx.now });

  // Apply break/leave requests made while the game was running, then re-queue.
  for (const id of everyone) {
    const p = playerById(n, id);
    if (p.pendingStatus) {
      p.status = p.pendingStatus;
      delete p.pendingStatus;
    }
    if (p.status === 'active') n.queueOrder.push(id);
  }

  if (court.closing) {
    n.courts = n.courts.filter((c) => c.id !== court.id);
  } else {
    court.status = 'empty';
    delete court.game;
  }

  fillCourts(n, ctx);
  return n;
}
```

```ts file=src/engine/courts.ts
import { cloneSession } from './clone';
import { EngineError } from './errors';
import { fillCourts } from './fill';
import type { Ctx, Court, Session } from './types';

export function addCourt(s: Session, ctx: Ctx): Session {
  const n = cloneSession(s);
  const number = Math.max(0, ...n.courts.map((c) => c.number)) + 1;
  const court: Court = { id: ctx.newId(), number, status: 'empty', closing: false };
  n.courts.push(court);
  fillCourts(n, ctx);
  return n;
}

/**
 * Empty court: removed immediately. Court in play: marked `closing` and removed
 * when its game ends. Calling again on a closing court cancels the closing.
 * At least one court must stay open.
 */
export function removeCourt(s: Session, courtId: string): Session {
  const n = cloneSession(s);
  const court = n.courts.find((c) => c.id === courtId);
  if (!court) throw new EngineError('NOT_FOUND', 'Court not found.');
  if (court.closing) {
    court.closing = false;
    return n;
  }
  if (n.courts.filter((c) => !c.closing).length <= 1) {
    throw new EngineError('LAST_COURT', 'At least one court must stay open.');
  }
  if (court.status === 'empty') {
    n.courts = n.courts.filter((c) => c.id !== court.id);
  } else {
    court.closing = true;
  }
  return n;
}
```

```ts file=src/engine/status.ts
import { cloneSession } from './clone';
import { EngineError } from './errors';
import { fillCourts } from './fill';
import { lateArrivalGames, onCourtIds, playerById } from './pool';
import type { Ctx, PlayerId, Session } from './types';

export type StatusTarget = 'active' | 'break' | 'left';

/**
 * Break / leave: immediate for waiting players; for a player on court it is
 * recorded as `pendingStatus` and applied when their game ends.
 * Active: returns a player from break (treated like a late arrival) or cancels
 * a pending break/leave. A player who left cannot return.
 */
export function setPlayerStatus(
  s: Session,
  playerId: PlayerId,
  target: StatusTarget,
  ctx: Ctx,
): Session {
  const n = cloneSession(s);
  const p = playerById(n, playerId);

  if (target === 'active') {
    if (p.status === 'left') {
      throw new EngineError('CANNOT_RETURN', `${p.name} has left the session.`);
    }
    if (p.pendingStatus) {
      delete p.pendingStatus;
      return n;
    }
    if (p.status === 'active') return n;
    p.status = 'active';
    p.gamesPlayed = Math.max(p.gamesPlayed, lateArrivalGames(n, p.id));
    p.lastFinishedSeq = n.gameSeq;
    n.queueOrder.push(p.id);
    fillCourts(n, ctx);
    return n;
  }

  if (p.status === 'left' || p.status === target) return n;

  if (onCourtIds(n).has(p.id)) {
    p.pendingStatus = target;
    return n;
  }

  p.status = target;
  n.queueOrder = n.queueOrder.filter((id) => id !== p.id);
  return n;
}
```

```ts file=src/engine/preview.ts
import { orderedWaiting } from './pool';
import type { PlayerId, Session } from './types';

export interface UpNext {
  courtNumber: number;
  playerIds: PlayerId[];
}

/**
 * Advisory preview of the next game: the 4 players first in line, and the court
 * most likely to free up first. It is recomputed for real when a court frees up.
 */
export function previewNext(s: Session): UpNext | null {
  const waiting = orderedWaiting(s);
  if (waiting.length < 4) return null;
  const open = s.courts.filter((c) => !c.closing);
  const empty = open.find((c) => c.status === 'empty');
  const soonest = open
    .filter((c) => c.status === 'playing' && c.game)
    .sort((a, b) => a.game!.startedAt - b.game!.startedAt)[0];
  const court = empty ?? soonest;
  if (!court) return null;
  return { courtNumber: court.number, playerIds: waiting.slice(0, 4).map((p) => p.id) };
}
```

```ts file=src/engine/summary.ts
import type { PlayerStatus, Session } from './types';

export interface PlayerSummary {
  id: string;
  name: string;
  gamesPlayed: number;
  status: PlayerStatus;
  partners: { name: string; count: number }[];
}

export function summarize(s: Session): PlayerSummary[] {
  const nameById = new Map(s.players.map((p) => [p.id, p.name]));
  return s.players
    .map((p) => ({
      id: p.id,
      name: p.name,
      gamesPlayed: p.gamesPlayed,
      status: p.status,
      partners: Object.entries(p.partnerCounts)
        .flatMap(([id, count]) => {
          const name = nameById.get(id);
          return name ? [{ name, count }] : [];
        })
        .sort((a, b) => b.count - a.count || a.name.localeCompare(b.name)),
    }))
    .sort((a, b) => b.gamesPlayed - a.gamesPlayed || a.name.localeCompare(b.name));
}

function plural(n: number, word: string): string {
  return `${n} ${word}${n === 1 ? '' : 's'}`;
}

export function formatSummaryText(s: Session): string {
  const date = new Date(s.createdAt).toISOString().slice(0, 10);
  const lines = [`${s.name} — ${date}`, `${plural(s.history.length, 'game')} played`, ''];
  for (const row of summarize(s)) {
    const partners = row.partners.length
      ? ` (${plural(row.partners.length, 'partner')}: ${row.partners
          .map((p) => (p.count > 1 ? `${p.name} ×${p.count}` : p.name))
          .join(', ')})`
      : '';
    lines.push(`${row.name}: ${plural(row.gamesPlayed, 'game')}${partners}`);
  }
  return lines.join('\n');
}
```

```ts file=src/engine/simulation.test.ts
function assertInvariants(s: Session): void {
  // queueOrder is exactly the set of available players, with no duplicates.
  const waiting = availablePlayers(s)
    .map((p) => p.id)
    .sort();
  expect([...s.queueOrder].sort()).toEqual(waiting);
  // Nobody is on two courts or twice in one game.
  const ids = s.courts.flatMap((c) => (c.game ? [...c.game.team1, ...c.game.team2] : []));
  expect(new Set(ids).size).toBe(ids.length);
  expect(onCourtIds(s).size).toBe(ids.length);
  // Idle courts only exist when fewer than 4 are waiting.
  if (s.courts.some((c) => c.status === 'empty')) expect(waiting.length).toBeLessThan(4);
}

function spread(s: Session): number {
  const games = s.players.filter((p) => p.status === 'active').map((p) => p.gamesPlayed);
  return Math.max(...games) - Math.min(...games);
}

    const base = makeCtx(42, 0);

      const playing = s.courts
        .filter((c) => c.status === 'playing')
        .sort((a, b) => a.game!.startedAt - b.game!.startedAt);
```

```ts file=src/store/persistence.ts
import type { Session } from '../engine';

export const STORAGE_KEY = 'pickleball.session.v1';
const VERSION = 1;

export interface Persisted {
  session: Session | null;
  undoStack: Session[];
}

function isSession(x: unknown): x is Session {
  if (typeof x !== 'object' || x === null) return false;
  const s = x as Record<string, unknown>;
  return (
    typeof s.id === 'string' &&
    typeof s.name === 'string' &&
    (s.mode === 'fairness' || s.mode === 'queue') &&
    Array.isArray(s.courts) &&
    Array.isArray(s.players) &&
    Array.isArray(s.queueOrder) &&
    Array.isArray(s.history) &&
    typeof s.gameSeq === 'number'
  );
}

export function loadPersisted(): Persisted | null {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return null;
    const data = JSON.parse(raw) as Record<string, unknown>;
    if (data.version !== VERSION) return null;
    if (data.session !== null && !isSession(data.session)) return null;
    if (!Array.isArray(data.undoStack) || !data.undoStack.every(isSession)) return null;
    return { session: data.session as Session | null, undoStack: data.undoStack as Session[] };
  } catch {
    return null;
  }
}

function write(p: Persisted): boolean {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({ version: VERSION, ...p }));
    return true;
  } catch {
    return false;
  }
}

/** Returns false if storage failed. Retries with a short undo stack if the quota is hit. */
export function savePersisted(p: Persisted): boolean {
  return write(p) || write({ session: p.session, undoStack: p.undoStack.slice(-5) });
}

export function clearPersisted(): void {
  try {
    localStorage.removeItem(STORAGE_KEY);
  } catch {
    /* nothing useful to do */
  }
}
```

```ts file=src/store/sessionStore.ts
import { create } from 'zustand';
import * as engine from '../engine';
import type { Ctx, NewSessionInput, Session } from '../engine';
import { liveCtx } from './liveCtx';
import { loadPersisted, savePersisted } from './persistence';

export const MAX_UNDO = 50;

export type SessionAction = (s: Session, ctx: Ctx) => Session;
export type ApplyResult = { ok: true } | { ok: false; error: string };

interface SessionState {
  session: Session | null;
  undoStack: Session[];
  /** True when the last attempt to save to the browser failed. */
  saveFailed: boolean;
  start: (input: NewSessionInput) => ApplyResult;
  apply: (action: SessionAction) => ApplyResult;
  undo: () => void;
  reset: () => void;
}

/** Engine errors are expected user-facing rejections; anything else is a bug and rethrown. */
function toResult(e: unknown): ApplyResult {
  if (e instanceof engine.EngineError) return { ok: false, error: e.message };
  throw e;
}

const initial = loadPersisted();

export const useSessionStore = create<SessionState>()((set, get) => {
  const commit = (session: Session | null, undoStack: Session[]) => {
    const saved = savePersisted({ session, undoStack });
    set({ session, undoStack, saveFailed: !saved });
  };

  return {
    session: initial?.session ?? null,
    undoStack: initial?.undoStack ?? [],
    saveFailed: false,

    start(input) {
      try {
        commit(engine.createSession(input, liveCtx()), []);
        return { ok: true };
      } catch (e) {
        return toResult(e);
      }
    },

    apply(action) {
      const { session, undoStack } = get();
      if (!session) return { ok: false, error: 'There is no active session.' };
      try {
        const next = action(session, liveCtx());
        commit(next, [...undoStack, session].slice(-MAX_UNDO));
        return { ok: true };
      } catch (e) {
        return toResult(e);
      }
    },

    undo() {
      const { undoStack } = get();
      const previous = undoStack[undoStack.length - 1];
      if (!previous) return;
      commit(previous, undoStack.slice(0, -1));
    },

    reset() {
      commit(null, []);
    },
  };
});
```

```ts file=src/store/rosterDb.ts
import { deleteDB, openDB, type DBSchema, type IDBPDatabase } from 'idb';
import { newId } from '../lib/id';

export interface RosterPlayer {
  id: string;
  name: string;
  /** 0 = no photo. Used by the photos plan. */
  photoVersion: number;
  lastPlayedAt?: number;
  /** Local edit time until synced (sync plan). */
  updatedAt: number;
  /** Soft delete so deletions can sync later. */
  deletedAt?: number;
}

interface RosterSchema extends DBSchema {
  players: { key: string; value: RosterPlayer };
}

const DB_NAME = 'pickleball-roster';
let dbPromise: Promise<IDBPDatabase<RosterSchema>> | null = null;

function db(): Promise<IDBPDatabase<RosterSchema>> {
  dbPromise ??= openDB<RosterSchema>(DB_NAME, 1, {
    upgrade(database) {
      database.createObjectStore('players', { keyPath: 'id' });
    },
  });
  return dbPromise;
}

/** Test helper: closes and deletes the database. */
export async function resetRosterDbForTests(): Promise<void> {
  if (dbPromise) (await dbPromise).close();
  dbPromise = null;
  await deleteDB(DB_NAME);
}

const norm = (name: string) => name.trim().toLowerCase();

async function liveEntries(): Promise<RosterPlayer[]> {
  return (await (await db()).getAll('players')).filter((p) => !p.deletedAt);
}

export async function listRoster(): Promise<RosterPlayer[]> {
  return (await liveEntries()).sort((a, b) =>
    a.name.localeCompare(b.name, undefined, { sensitivity: 'base' }),
  );
}

/** Adds a player, or returns the existing entry if the name is already saved. */
export async function addRosterPlayer(name: string, now = Date.now()): Promise<RosterPlayer> {
  const clean = name.trim();
  if (!clean) throw new Error('Enter a name for the player.');
  const existing = (await liveEntries()).find((p) => norm(p.name) === norm(clean));
  if (existing) return existing;
  const player: RosterPlayer = { id: newId(), name: clean, photoVersion: 0, updatedAt: now };
  await (await db()).put('players', player);
  return player;
}

export async function renameRosterPlayer(id: string, name: string, now = Date.now()): Promise<void> {
  const clean = name.trim();
  if (!clean) throw new Error('Enter a name for the player.');
  const entries = await liveEntries();
  if (entries.some((p) => p.id !== id && norm(p.name) === norm(clean))) {
    throw new Error(`${clean} is already saved.`);
  }
  const current = entries.find((p) => p.id === id);
  if (!current) throw new Error('That player no longer exists.');
  await (await db()).put('players', { ...current, name: clean, updatedAt: now });
}

export async function deleteRosterPlayer(id: string, now = Date.now()): Promise<void> {
  const database = await db();
  const current = await database.get('players', id);
  if (!current || current.deletedAt) return;
  await database.put('players', { ...current, deletedAt: now, updatedAt: now });
}

export async function markRosterPlayed(ids: string[], at: number): Promise<void> {
  const database = await db();
  for (const id of ids) {
    const current = await database.get('players', id);
    if (current && !current.deletedAt) {
      await database.put('players', { ...current, lastPlayedAt: at, updatedAt: at });
    }
  }
}

/** The players of the most recent session: everyone sharing the latest lastPlayedAt. */
export function lastSessionIds(roster: RosterPlayer[]): string[] {
  const times = roster.flatMap((p) => (p.lastPlayedAt === undefined ? [] : [p.lastPlayedAt]));
  if (times.length === 0) return [];
  const latest = Math.max(...times);
  return roster.filter((p) => p.lastPlayedAt === latest).map((p) => p.id);
}
```

```ts file=src/lib/time.ts
/** Formats a duration as m:ss, or h:mm:ss from one hour up. */
export function formatElapsed(ms: number): string {
  const total = Math.max(0, Math.floor(ms / 1000));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  const ss = String(s).padStart(2, '0');
  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${ss}`;
  return `${m}:${ss}`;
}
```

```ts file=src/ui/theme.ts
export type Theme = 'light' | 'dark';

const KEY = 'pickleball.theme';

export function getInitialTheme(): Theme {
  try {
    const saved = localStorage.getItem(KEY);
    if (saved === 'light' || saved === 'dark') return saved;
  } catch {
    /* storage unavailable: fall through to the system preference */
  }
  return window.matchMedia?.('(prefers-color-scheme: dark)')?.matches ? 'dark' : 'light';
}

/** Applies the theme to the page without remembering it. */
export function applyTheme(theme: Theme): void {
  document.documentElement.dataset.theme = theme;
}

/** Applies the theme and remembers the user's choice. */
export function setTheme(theme: Theme): void {
  applyTheme(theme);
  try {
    localStorage.setItem(KEY, theme);
  } catch {
    /* the choice just won't persist */
  }
}
```

```tsx file=src/ui/Avatar.tsx
export function initials(name: string): string {
  const parts = name.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return '?';
  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
}

  }
  return (
    <span className={`avatar avatar-${size}`} aria-hidden="true">

    </span>
  );
}
```

```tsx file=src/ui/SetupScreen.tsx
const MIN_PLAYERS = 4;
const MAX_COURTS = 12;

interface Props {
  onManage: () => void;
}

export function SetupScreen({ onManage }: Props) {
  const roster = useRosterStore((s) => s.players);
  const addToRoster = useRosterStore((s) => s.add);
  const start = useSessionStore((s) => s.start);

  const [name, setName] = useState('');
  const [courtCount, setCourtCount] = useState(2);
  const [mode, setMode] = useState<Mode>('fairness');
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [draft, setDraft] = useState('');
  const [error, setError] = useState<string | null>(null);

  const lastIds = lastSessionIds(roster);
  const chosen = roster.filter((p) => selected.has(p.id));
  const canStart = chosen.length >= MIN_PLAYERS;

  function toggle(id: string) {
    setSelected((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  }

  async function addPlayers() {
    const lines = draft
      .split('\n')
      .map((l) => l.trim())
      .filter(Boolean);
    if (lines.length === 0) return;
    const ids: string[] = [];
    let failure: string | null = null;
    for (const line of lines) {
      try {
        ids.push((await addToRoster(line)).id);
      } catch (e) {
        failure = e instanceof Error ? e.message : 'Could not add that player.';
      }
    }
    setSelected((prev) => new Set([...prev, ...ids]));
    setDraft('');
    setError(failure);
  }

  function startSession() {
    const res = start({
      name,
      mode,
      courtCount,
      players: chosen.map((p) => ({ name: p.name, rosterId: p.id })),

    if (!res.ok) setError(res.error);
  }

  return (
    <div className="stack">
      <h1 className="screen-title">New session</h1>

      <section className="card stack">
```

```tsx file=src/ui/LiveScreen.tsx
    () => new Map(session.players.map((p) => [p.id, p])),
    [session.players],
  );

  function run(action: Parameters<typeof apply>[0]) {
    const res = apply(action);
    setError(res.ok ? null : res.error);
  }

  function endSession() {
    if (!window.confirm('End this session? You can start a new one afterwards.')) return;
    const rosterIds = session.players.flatMap((p) => (p.rosterId ? [p.rosterId] : []));
    void markPlayed(rosterIds, Date.now());
    run((s, ctx) => engine.endSession(s, ctx));
  }

  const sorted = [...session.courts].sort((a, b) => a.number - b.number);

  return (
    <div>
      <div className="row-flex spread">
        <h1 className="screen-title">{session.name}</h1>
        <span className="muted">
          {session.history.length} {session.history.length === 1 ? 'game' : 'games'} played
        </span>
      </div>

      {error && (
        <p className="error" role="alert">
          {error}
        </p>
      )}

      <div className="live-grid">
        <div className="courts">
          {sorted.map((court) => (
            <CourtCard
              key={court.id}
              court={court}
              playersById={playersById}
              onFinish={() => run((s, ctx) => engine.finishGame(s, court.id, ctx))}
              onToggleClosing={() => run((s) => engine.removeCourt(s, court.id))}
              onSelectPlayer={setSelectedId}
            />
          ))}
        </div>

        <div className="stack">
          <UpNextCard session={session} />
          <WaitingList session={session} onSelect={setSelectedId} onAdd={() => setAdding(true)} />
        </div>
      </div>

      <div className="controls">
        <button type="button" className="btn" disabled={undoCount === 0} onClick={undo}>
          Undo
        </button>
        <button type="button" className="btn" onClick={() => run((s, ctx) => engine.addCourt(s, ctx))}>
          Add court
        </button>
        <button type="button" className="btn btn-danger" onClick={endSession}>
          End session
        </button>
      </div>

      {selectedId && (
        <PlayerSheet session={session} playerId={selectedId} onClose={() => setSelectedId(null)} />
      )}
      {adding && <AddPlayerSheet session={session} onClose={() => setAdding(false)} />}
    </div>
  );
}
```

```js file=scripts/make-icons.mjs
import { mkdirSync, writeFileSync } from 'node:fs';
import { deflateSync } from 'node:zlib';

const LIME = [200, 240, 60];
const INK = [17, 17, 17];

const crcTable = new Uint32Array(256).map((_, n) => {
  let c = n;
  for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
  return c >>> 0;
});

function crc32(buf) {
  let c = 0xffffffff;
  for (const b of buf) c = crcTable[(c ^ b) & 0xff] ^ (c >>> 8);
  return (c ^ 0xffffffff) >>> 0;
}

function chunk(type, data) {
  const len = Buffer.alloc(4);
  len.writeUInt32BE(data.length);
  const body = Buffer.concat([Buffer.from(type), data]);
  const crc = Buffer.alloc(4);
  crc.writeUInt32BE(crc32(body));
  return Buffer.concat([len, body, crc]);
}

function pixel(x, y, size) {
  const cx = size / 2;
  const cy = size / 2;
  const dist = (px, py) => Math.hypot(x - px, y - py);
  const ball = size * 0.3;
  if (dist(cx, cy) <= ball) {
    const holes = [
      [cx - ball * 0.4, cy - ball * 0.35],
      [cx + ball * 0.35, cy - ball * 0.45],
      [cx + ball * 0.5, cy + ball * 0.15],
      [cx - ball * 0.15, cy + ball * 0.5],
      [cx - ball * 0.6, cy + ball * 0.1],
    ];
    if (holes.some(([hx, hy]) => dist(hx, hy) <= size * 0.045)) return LIME;
    return INK;
  }
  return LIME;
}

function png(size) {
  const stride = size * 3 + 1;
  const raw = Buffer.alloc(stride * size);
  for (let y = 0; y < size; y++) {
    raw[y * stride] = 0; // filter: none
    for (let x = 0; x < size; x++) {
      const [r, g, b] = pixel(x, y, size);
      const i = y * stride + 1 + x * 3;
      raw[i] = r;
      raw[i + 1] = g;
      raw[i + 2] = b;
    }
  }
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(size, 0);
  ihdr.writeUInt32BE(size, 4);
  ihdr[8] = 8; // bit depth
  ihdr[9] = 2; // colour type: RGB
  return Buffer.concat([
    Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]),
    chunk('IHDR', ihdr),
    chunk('IDAT', deflateSync(raw)),
    chunk('IEND', Buffer.alloc(0)),
  ]);
}

mkdirSync('public/icons', { recursive: true });
for (const size of [192, 512]) {
  writeFileSync(`public/icons/icon-${size}.png`, png(size));
  console.log(`wrote public/icons/icon-${size}.png`);
}
```

````md file=README.md
# Pickleball Matchmaker

Automatic matchmaking for pickleball open play. Create a session, set the number of courts, add players, and the app decides who plays on which court and who goes next when a game ends. Works on phones and desktops, installs as an app, and works offline.
````
